import { useState, useRef, useEffect } from "react";
import { Menu, LogOut, UserCircle, Settings, ChevronDown } from "lucide-react";
import { useNavigate } from "react-router-dom";
import { useAuth } from "../../context/AuthContext";
import { APP_NAME, ORG_NAME } from "../../utils/constants";

function Navbar({ onMenuClick }) {
  const { user, logout } = useAuth();
  const navigate = useNavigate();

  const [isProfileMenuOpen, setIsProfileMenuOpen] = useState(false);
  const profileMenuRef = useRef(null);

  // Close the dropdown when clicking anywhere outside it
  useEffect(() => {
    function handleClickOutside(e) {
      if (profileMenuRef.current && !profileMenuRef.current.contains(e.target)) {
        setIsProfileMenuOpen(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const handleLogout = () => {
    logout();
    navigate("/");
  };

  const goToProfile = () => {
    setIsProfileMenuOpen(false);
    navigate("/app/profile");
  };

  const goToSettings = () => {
    setIsProfileMenuOpen(false);
    navigate("/app/settings");
  };

  return (
    <header className="navbar">
      <div className="navbar-left">
        <button className="navbar-menu-btn" onClick={onMenuClick} aria-label="Toggle sidebar" title="Toggle sidebar">
          <Menu size={22} />
        </button>
        <div className="navbar-title">
          <span className="navbar-org">{ORG_NAME}</span>
          <span className="navbar-app-name">{APP_NAME}</span>
        </div>
      </div>

      <div className="navbar-right">
        <div className="navbar-profile-wrapper" ref={profileMenuRef}>
          <button
            className="navbar-user"
            onClick={() => setIsProfileMenuOpen((prev) => !prev)}
            aria-label="Open profile menu"
          >
            <UserCircle size={22} />
            <span className="navbar-username">{user?.fullName || "User"}</span>
            <ChevronDown size={14} />
          </button>

          {isProfileMenuOpen && (
            <div className="profile-dropdown">
              <div className="profile-dropdown-header">
                <p className="profile-dropdown-name">{user?.fullName || "User"}</p>
                <p className="profile-dropdown-email">{user?.email || ""}</p>
              </div>
              <button className="profile-dropdown-item" onClick={goToProfile}>
                <UserCircle size={16} /> View Profile
              </button>
              <button className="profile-dropdown-item" onClick={goToSettings}>
                <Settings size={16} /> Settings
              </button>
              <button className="profile-dropdown-item profile-dropdown-logout" onClick={handleLogout}>
                <LogOut size={16} /> Logout
              </button>
            </div>
          )}
        </div>
      </div>
    </header>
  );
}

import { Marker, Popup } from "react-leaflet";
import L from "leaflet";
import { CONFIDENCE_LEVELS } from "../../utils/constants";

function getMarkerIcon(confidence) {
  let color = "#dc2626";
  if (confidence >= CONFIDENCE_LEVELS.HIGH) color = "#16a34a";
  else if (confidence >= CONFIDENCE_LEVELS.MEDIUM) color = "#f59e0b";

  return L.divIcon({
    className: "custom-well-marker",
    html: `<div style="
      width: 16px;
      height: 16px;
      background-color: ${color};
      border: 2px solid #ffffff;
      border-radius: 50%;
      box-shadow: 0 0 4px rgba(0,0,0,0.4);
    "></div>`,
    iconSize: [16, 16],
    iconAnchor: [8, 8],
    popupAnchor: [0, -8],
  });
}

function WellMarker({ well, onViewDetails }) {
  return (
    <Marker position={[well.latitude, well.longitude]} icon={getMarkerIcon(well.confidence)}>
      <Popup>
        <div className="well-popup">
          <p><strong>Well ID:</strong> {well.id}</p>
          <p><strong>Latitude:</strong> {well.latitude}</p>
          <p><strong>Longitude:</strong> {well.longitude}</p>
          <p><strong>Confidence:</strong> {Math.round(well.confidence * 100)}%</p>
          <p><strong>Status:</strong> {well.status}</p>
          <p><strong>Village:</strong> {well.village}</p>
          {onViewDetails && (
            <button className="popup-details-btn" onClick={() => onViewDetails(well)}>
              View Details
            </button>
          )}
        </div>
      </Popup>
    </Marker>
  );
}

// src/components/wells/WellTable.jsx
import { useState, useMemo } from "react";
import { ArrowUpDown } from "lucide-react";

function WellTable({ wells, onRowClick }) {
  const [searchTerm, setSearchTerm] = useState("");
  const [statusFilter, setStatusFilter] = useState("All");
  const [sortConfig, setSortConfig] = useState({ key: "id", direction: "asc" });

  const handleSort = (key) => {
    setSortConfig((prev) => ({
      key,
      direction: prev.key === key && prev.direction === "asc" ? "desc" : "asc",
    }));
  };

  const filteredAndSortedWells = useMemo(() => {
    let result = wells.filter((well) => {
      const matchesSearch =
        well.id.toLowerCase().includes(searchTerm.toLowerCase()) ||
        well.village.toLowerCase().includes(searchTerm.toLowerCase());
      const matchesStatus = statusFilter === "All" || well.status === statusFilter;
      return matchesSearch && matchesStatus;
    });

    result.sort((a, b) => {
      const valA = a[sortConfig.key];
      const valB = b[sortConfig.key];
      if (valA < valB) return sortConfig.direction === "asc" ? -1 : 1;
      if (valA > valB) return sortConfig.direction === "asc" ? 1 : -1;
      return 0;
    });

    return result;
  }, [wells, searchTerm, statusFilter, sortConfig]);

  const statusOptions = ["All", ...new Set(wells.map((w) => w.status))];

  return (
    <div className="well-table-wrapper">
      <div className="table-controls">
        <input
          type="text"
          className="table-search-input"
          placeholder="Search by Well ID or Village..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
        />
        <select
          className="table-filter-select"
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value)}
        >
          {statusOptions.map((status) => (
            <option key={status} value={status}>{status}</option>
          ))}
        </select>
      </div>

      <div className="table-scroll-container">
        <table className="wells-table">
          <thead>
            <tr>
              <th onClick={() => handleSort("id")}>Well ID <ArrowUpDown size={12} /></th>
              <th onClick={() => handleSort("latitude")}>Latitude <ArrowUpDown size={12} /></th>
              <th onClick={() => handleSort("longitude")}>Longitude <ArrowUpDown size={12} /></th>
              <th onClick={() => handleSort("confidence")}>Confidence <ArrowUpDown size={12} /></th>
              <th onClick={() => handleSort("status")}>Status <ArrowUpDown size={12} /></th>
            </tr>
          </thead>
          <tbody>
            {filteredAndSortedWells.length === 0 ? (
              <tr>
                <td colSpan={5} className="table-empty-row">No wells match your search/filter.</td>
              </tr>
            ) : (
              filteredAndSortedWells.map((well) => (
                <tr key={well.id} onClick={() => onRowClick(well)} className="table-row-clickable">
                  <td>{well.id}</td>
                  <td>{well.latitude}</td>
                  <td>{well.longitude}</td>
                  <td>{Math.round(well.confidence * 100)}%</td>
                  <td><span className="status-pill">{well.status}</span></td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}

// src/context/AuthContext.jsx
import { createContext, useContext, useState, useEffect } from "react";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  // On app load, check if a user session was saved (mock persistence via localStorage)
  useEffect(() => {
    const savedUser = localStorage.getItem("agriWellUser");
    if (savedUser) {
      setUser(JSON.parse(savedUser));
    }
    setLoading(false);
  }, []);

  const login = (userData) => {
    setUser(userData);
    localStorage.setItem("agriWellUser", JSON.stringify(userData));
  };

  const logout = () => {
    setUser(null);
    localStorage.removeItem("agriWellUser");
  };

  const value = {
    user,
    isAuthenticated: !!user,
    loading,
    login,
    logout,
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

// Custom hook for easy access: const { user, login, logout } = useAuth();
export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used within an AuthProvider");
  }
  return context;
}

import { FileText, Download, FileSpreadsheet } from "lucide-react";
import { getReportSummary } from "../services/reportService";
import { getAllWells } from "../services/wellService";
import { exportWellsToCSV, generateReport } from "../services/exportService";
import WellMap from "../components/map/WellMap";
import Loader from "../components/common/Loader";
import ErrorMessage from "../components/common/ErrorMessage";

function Reports() {
  const [summary, setSummary] = useState(null);
  const [wells, setWells] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState("");

  const [isGenerating, setIsGenerating] = useState(false);
  const [actionMessage, setActionMessage] = useState("");

  useEffect(() => {
    async function loadReportData() {
      setIsLoading(true);
      setError("");
      try {
        const [summaryResult, wellsResult] = await Promise.all([
          getReportSummary(),
          getAllWells(),
        ]);
        setSummary(summaryResult);
        setWells(wellsResult);
      } catch (err) {
        setError("Unable to load report data. Please try again later.");
      } finally {
        setIsLoading(false);
      }
    }
    loadReportData();
  }, []);

  if (isLoading) return <Loader message="Loading report data..." />;
  if (error) return <ErrorMessage message={error} />;

  const handleGenerateReport = async () => {
    setIsGenerating(true);
    setActionMessage("");
    try {
      const report = await generateReport(wells);
      setActionMessage(`Report ${report.reportId} generated successfully.`);
    } catch (err) {
      setActionMessage(err.message || "Failed to generate report.");
    } finally {
      setIsGenerating(false);
      setTimeout(() => setActionMessage(""), 4000);
    }
  };

  const handleDownloadPDF = () => {
    // Mock — real PDF generation requires backend rendering (or a PDF lib later)
    setActionMessage("PDF download is not yet connected to a backend. This is a placeholder.");
    setTimeout(() => setActionMessage(""), 4000);
  };

  const handleDownloadCSV = () => {
    try {
      exportWellsToCSV(wells, "well-detection-report.csv");
      setActionMessage("CSV downloaded successfully.");
    } catch (err) {
      setActionMessage(err.message);
    }
    setTimeout(() => setActionMessage(""), 3000);
  };

  return (
    <div className="reports-page">
      <h2 className="page-heading">Reports</h2>
      <p className="page-subheading">Generate and download detection reports</p>

      {actionMessage && <div className="action-message-banner">{actionMessage}</div>}

      <div className="report-summary-card">
        <h3 className="section-title">Detection Summary</h3>
        <div className="report-summary-grid">
          <div className="report-summary-item">
            <span className="detail-label">Total Wells</span>
            <span className="detail-value">{summary.totalWells}</span>
          </div>
          <div className="report-summary-item">
            <span className="detail-label">High Confidence</span>
            <span className="detail-value">{summary.highConfidence}</span>
          </div>
          <div className="report-summary-item">
            <span className="detail-label">Medium Confidence</span>
            <span className="detail-value">{summary.mediumConfidence}</span>
          </div>
          <div className="report-summary-item">
            <span className="detail-label">Low Confidence</span>
            <span className="detail-value">{summary.lowConfidence}</span>
          </div>
          <div className="report-summary-item">
            <span className="detail-label">Area(s) Covered</span>
            <span className="detail-value">{summary.areasCovered}</span>
          </div>
          <div className="report-summary-item">
            <span className="detail-label">Latest Detection Date</span>
            <span className="detail-value">{summary.latestDetectionDate}</span>
          </div>
        </div>

      </div>
    </div>
  );
}

// src/pages/Results.jsx
import { useEffect, useState } from "react";
import { Download, FileText, Map as MapIcon } from "lucide-react";
import { getAllWells } from "../services/wellService";
import { exportWellsToCSV, generateReport } from "../services/exportService";
import { CONFIDENCE_LEVELS } from "../utils/constants";
import WellTable from "../components/wells/WellTable";
import WellDetails from "../components/wells/WellDetails";
import WellMap from "../components/map/WellMap";
import Loader from "../components/common/Loader";
import ErrorMessage from "../components/common/ErrorMessage";

function Results() {
  const [wells, setWells] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState("");

  const [selectedWell, setSelectedWell] = useState(null);
  const [showMap, setShowMap] = useState(false);

  const [isGeneratingReport, setIsGeneratingReport] = useState(false);
  const [actionMessage, setActionMessage] = useState("");

  useEffect(() => {
    async function loadWells() {
      setIsLoading(true);
      setError("");
      try {
        const data = await getAllWells();
        setWells(data);
      } catch (err) {
        setError("Unable to load detection results. Please try again later.");
      } finally {
        setIsLoading(false);
      }
    }
    loadWells();
  }, []);

  if (isLoading) return <Loader message="Loading results..." />;
  if (error) return <ErrorMessage message={error} />;

  // Summary counts derived from the loaded wells
  const totalWells = wells.length;
  const highConfidence = wells.filter((w) => w.confidence >= CONFIDENCE_LEVELS.HIGH).length;
  const mediumConfidence = wells.filter(
    (w) => w.confidence >= CONFIDENCE_LEVELS.MEDIUM && w.confidence < CONFIDENCE_LEVELS.HIGH
  ).length;
  const lowConfidence = wells.filter((w) => w.confidence < CONFIDENCE_LEVELS.MEDIUM).length;

  const summaryCards = [
    { label: "Total Wells", value: totalWells, color: "#14532d" },
    { label: "High Confidence", value: highConfidence, color: "#16a34a" },
    { label: "Medium Confidence", value: mediumConfidence, color: "#f59e0b" },
    { label: "Low Confidence", value: lowConfidence, color: "#dc2626" },
  ];

  const handleDownloadCSV = () => {
    try {
      exportWellsToCSV(wells);
      setActionMessage("CSV downloaded successfully.");
    } catch (err) {
      setActionMessage(err.message);
    }
    setTimeout(() => setActionMessage(""), 3000);
  };

  const handleGenerateReport = async () => {
    setIsGeneratingReport(true);
    setActionMessage("");
    try {
      const report = await generateReport(wells);
      setActionMessage(`Report ${report.reportId} generated successfully.`);
    } catch (err) {
      setActionMessage(err.message || "Failed to generate report.");
    } finally {
      setIsGeneratingReport(false);
      setTimeout(() => setActionMessage(""), 4000);
    }
  };

  return (
    <div className="results-page">
      <h2 className="page-heading">Detection Results</h2>
      <p className="page-subheading">Full breakdown of all detected wells</p>

      {actionMessage && <div className="action-message-banner">{actionMessage}</div>}

      <div className="stat-cards-grid">
        {summaryCards.map(({ label, value, color }) => (
          <div className="stat-card" key={label}>
            <div>
              <p className="stat-value" style={{ color }}>{value}</p>
              <p className="stat-label">{label}</p>
            </div>
          </div>
        ))}
      </div>

      <div className="results-actions-row">
        <button className="secondary-btn" onClick={() => setShowMap((prev) => !prev)}>
          <MapIcon size={16} /> {showMap ? "Hide Map" : "View Map"}
        </button>
        <button className="secondary-btn" onClick={handleDownloadCSV}>
          <Download size={16} /> Download CSV
        </button>
        <button className="start-detection-btn" onClick={handleGenerateReport} disabled={isGeneratingReport}>
          <FileText size={16} /> {isGeneratingReport ? "Generating..." : "Generate Report"}
        </button>
      </div>

      {showMap && (
        <div className="results-map-block">
          <WellMap wells={wells} />
        </div>
      )}

      <WellTable wells={wells} onRowClick={setSelectedWell} />

      {selectedWell && (
        <WellDetails well={selectedWell} onClose={() => setSelectedWell(null)} />
      )}
    </div>
  );
}

import { useState } from "react";
import { Search, PlayCircle } from "lucide-react";
import { runDetection } from "../services/detectionService";
import WellMap from "../components/map/WellMap";
import Loader from "../components/common/Loader";
import ErrorMessage from "../components/common/ErrorMessage";
import { useNavigate } from "react-router-dom";

// Workflow stages, matches the spec's step list
const STAGES = {
  SELECT: "select",
  PREVIEW: "preview",
  PROCESSING: "processing",
  COMPLETED: "completed",
};

const AREA_TYPES = ["Village", "Taluka", "District", "Custom Area"];

function WellDetection() {
  const navigate = useNavigate();

  const [areaType, setAreaType] = useState("District");
  const [formData, setFormData] = useState({
    district: "",
    taluka: "",
    village: "",
    latitude: "",
    longitude: "",
  });

  const [stage, setStage] = useState(STAGES.SELECT);
  const [error, setError] = useState("");
  const [detectedWells, setDetectedWells] = useState([]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
  };

  const handlePreview = (e) => {
    e.preventDefault();
    setError("");

    if (!formData.district.trim()) {
      setError("Please enter at least a District to preview the area.");
      return;
    }

    setStage(STAGES.PREVIEW);
  };

  const handleStartDetection = async () => {
    setError("");
    setStage(STAGES.PROCESSING);

    try {
      const results = await runDetection(formData);
      setDetectedWells(results);
      setStage(STAGES.COMPLETED);
    } catch (err) {
      setError(err.message || "Detection failed. Please try again.");
      setStage(STAGES.PREVIEW); // fall back so user can retry
    }
  };

  const handleReset = () => {
    setStage(STAGES.SELECT);
    setDetectedWells([]);
    setError("");
  };

  return (
    <div className="detection-page">
      <h2 className="page-heading">Agricultural Well Detection</h2>
      <p className="page-subheading">Select an area to run automated well detection</p>

      {error && <ErrorMessage message={error} />}

      {/* ---- Stage: SELECT ---- */}
      {stage === STAGES.SELECT && (
        <div className="detection-form-card">
          <div className="area-type-tabs">
            {AREA_TYPES.map((type) => (
              <button
                key={type}
                type="button"
                className={`area-type-tab ${areaType === type ? "area-type-tab-active" : ""}`}
                onClick={() => setAreaType(type)}
              >
                {type}
              </button>
            ))}
          </div>

          <form onSubmit={handlePreview} className="detection-form">
            <div className="form-row">
              <div className="form-group">
                <label htmlFor="district">District</label>
                <input
                  id="district"
                  name="district"
                  type="text"
                  placeholder="e.g. Nagpur"
                  value={formData.district}
                  onChange={handleChange}
                />
              </div>
              <div className="form-group">
                <label htmlFor="taluka">Taluka</label>
                <input
                  id="taluka"
                  name="taluka"
                  type="text"
                  placeholder="e.g. Kalmeshwar"
                  value={formData.taluka}
                  onChange={handleChange}
                />
              </div>
              <div className="form-group">
                <label htmlFor="village">Village</label>
                <input
                  id="village"
                  name="village"
                  type="text"
                  placeholder="e.g. Wadi"
                  value={formData.village}
                  onChange={handleChange}
                />
              </div>
            </div>

            <div className="form-row">
              <div className="form-group">
                <label htmlFor="latitude">Latitude</label>
                <input
                  id="latitude"
                  name="latitude"
                  type="text"
                  placeholder="21.1458"
                  value={formData.latitude}
                  onChange={handleChange}
                />
              </div>
              <div className="form-group">
                <label htmlFor="longitude">Longitude</label>
                <input
                  id="longitude"
                  name="longitude"
                  type="text"
                  placeholder="79.0882"
                  value={formData.longitude}
                  onChange={handleChange}
                />
              </div>
              <div className="form-group">
                <label>&nbsp;</label>
                <button type="button" className="search-location-btn">
                  <Search size={16} /> Search Location
                </button>
              </div>
            </div>

            <button type="submit" className="preview-btn">
              Preview Area
            </button>
          </form>
        </div>
      )}

      {/* ---- Stage: PREVIEW ---- */}
      {stage === STAGES.PREVIEW && (
        <div className="detection-preview-card">
          <h3 className="section-title">Area Preview</h3>
          <div className="preview-summary">
            <p><strong>District:</strong> {formData.district}</p>
            {formData.taluka && <p><strong>Taluka:</strong> {formData.taluka}</p>}
            {formData.village && <p><strong>Village:</strong> {formData.village}</p>}
            {formData.latitude && <p><strong>Latitude:</strong> {formData.latitude}</p>}
            {formData.longitude && <p><strong>Longitude:</strong> {formData.longitude}</p>}
          </div>

            <button className="secondary-btn" onClick={handleReset}>

            </button>

              <PlayCircle size={18} /> Start Well Detection
            </button>
          </div>
        </div>
      )}

      {/* ---- Stage: PROCESSING ---- */}
      {stage === STAGES.PROCESSING && (
        <div className="detection-processing-card">
          <Loader message="Running detection on the selected area..." />
        </div>
      )}

      {/* ---- Stage: COMPLETED ---- */}
      {stage === STAGES.COMPLETED && (
        <div className="detection-completed-card">
          <h3 className="section-title">Detection Completed</h3>

          {detectedWells.length === 0 ? (
            <>

                Try a Different Area
              </button>

          ) : (
            <>

                  View Full Results
                </button>

          )}

      )}
    </div>
  );
}

// src/pages/auth/Login.jsx
import { useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import { Eye, EyeOff } from "lucide-react";
import { useAuth } from "../../context/AuthContext";
import { loginUser } from "../../services/authService";
import { isValidEmail } from "../../utils/validators";
import { APP_NAME, ORG_NAME } from "../../utils/constants";

function Login() {
  const navigate = useNavigate();
  const { login } = useAuth();

  const [formData, setFormData] = useState({ email: "", password: "" });
  const [rememberMe, setRememberMe] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [errors, setErrors] = useState({});
  const [serverError, setServerError] = useState("");
  const [isLoading, setIsLoading] = useState(false);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
    setErrors((prev) => ({ ...prev, [name]: "" })); // clear error as user types
  };

  const validate = () => {
    const newErrors = {};
    if (!formData.email) newErrors.email = "Email is required.";
    else if (!isValidEmail(formData.email)) newErrors.email = "Enter a valid email.";

    if (!formData.password) newErrors.password = "Password is required.";

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setServerError("");

    if (!validate()) return;

    setIsLoading(true);
    try {
      const user = await loginUser(formData);
      login(user); // saves to AuthContext + localStorage
      navigate("/app/dashboard");
    } catch (err) {
      // Friendly message only — never expose raw backend errors
      setServerError(err.message || "Login failed. Please try again.");
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="auth-page">
      <div className="auth-card">
        <div className="auth-header">
          <Link to="/" style={{ textDecoration: "none" }}>
            <h2>{ORG_NAME}</h2>
          </Link>
          <p>{APP_NAME}</p>
        </div>

        <h3 className="auth-form-title">Login</h3>

        {serverError && <div className="auth-error-banner">{serverError}</div>}

        <form onSubmit={handleSubmit} noValidate>
          <div className="form-group">
            <label htmlFor="email">Email</label>
            <input
              id="email"
              name="email"
              type="email"
              value={formData.email}
              onChange={handleChange}

function ProtectedRoute({ children }) {
  const { isAuthenticated, loading } = useAuth();

  if (!isAuthenticated) return <Navigate to="/login" replace />;

  return children;

// Centralized Axios instance.
// All service files (authService, wellService, detectionService, reportService)
// should import and use THIS instance instead of calling axios directly.
// This is the ONLY place that needs to change if the backend's base URL,
// auth token handling, or response format changes.

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || "http://localhost:8000/api",
  timeout: 15000,
  headers: {
    "Content-Type": "application/json",
  },
});

// ----- Request Interceptor -----
// Attach auth token automatically if one exists (e.g. after real login).
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem("agriWellToken");
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// ----- Response Interceptor -----
// Normalizes errors so components never see raw axios/network errors —
// they only ever see a clean `.message` string (per spec: never expose raw backend errors).
api.interceptors.response.use(
  (response) => response,
  (error) => {
    let friendlyMessage = "Something went wrong. Please try again.";

    if (error.response) {
      // Backend responded with an error status (4xx/5xx)
      friendlyMessage =
        error.response.data?.message || `Request failed (${error.response.status}).`;
    } else if (error.request) {
      // Request was made but no response received (network/server down)
      friendlyMessage = "Unable to reach the server. Please check your connection.";
    }

    return Promise.reject(new Error(friendlyMessage));
  }
);

/**
 * Converts an array of well objects into a CSV string and triggers a download.
 * This is a genuine working CSV export (not mocked) since it's pure frontend logic —
 * no backend is needed for this specific feature.
 */
export function exportWellsToCSV(wells, filename = "well-detection-results.csv") {
  if (!wells || wells.length === 0) {
    throw new Error("No data available to export.");
  }

  const headers = ["Well ID", "Latitude", "Longitude", "Confidence", "Status", "Village", "District", "Detected On"];
  const rows = wells.map((w) => [
    w.id,
    w.latitude,
    w.longitude,
    `${Math.round(w.confidence * 100)}%`,
    w.status,
    w.village,
    w.district,
    w.detectedOn,
  ]);

  const csvContent = [headers, ...rows].map((row) => row.join(",")).join("\n");
  const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);

  const link = document.createElement("a");
  link.href = url;
  link.setAttribute("download", filename);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  URL.revokeObjectURL(url);
}

// src/services/reportService.js
import mockWells from "../data/mockWells";
import { CONFIDENCE_LEVELS } from "../utils/constants";

const MOCK_DELAY = 500;
function delay(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/**
 * Fetch data needed for the Reports page summary.
 * Later replace with:
 *   const res = await api.get("/reports/summary");
 *   return res.data.data;
 */
export async function getReportSummary() {
  await delay(MOCK_DELAY);

  const totalWells = mockWells.length;
  const high = mockWells.filter((w) => w.confidence >= CONFIDENCE_LEVELS.HIGH).length;
  const medium = mockWells.filter(
    (w) => w.confidence >= CONFIDENCE_LEVELS.MEDIUM && w.confidence < CONFIDENCE_LEVELS.HIGH
  ).length;
  const low = mockWells.filter((w) => w.confidence < CONFIDENCE_LEVELS.MEDIUM).length;

  const districts = [...new Set(mockWells.map((w) => w.district))];
  const latestDetectionDate = mockWells
    .map((w) => w.detectedOn)
    .sort()
    .reverse()[0];

  return {
    totalWells,
    highConfidence: high,
    mediumConfidence: medium,
    lowConfidence: low,
    areasCovered: districts.join(", "),
    latestDetectionDate,
  };
}

// src/services/wellService.js
import mockWells from "../data/mockWells";
import { CONFIDENCE_LEVELS } from "../utils/constants";

const MOCK_DELAY = 500;

function delay(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/**
 * Fetch all detected wells.
 * Later replace with:
 *   const res = await api.get("/wells");
 *   return res.data.data;
 */
export async function getAllWells() {
  await delay(MOCK_DELAY);
  return mockWells;
}

/**
 * Fetch summary stats for the dashboard.
 * Later this may become its own backend endpoint (e.g. GET /wells/summary)
 * instead of being computed on the frontend.
 */
export async function getDashboardStats() {
  await delay(MOCK_DELAY);

  const totalWells = mockWells.length;
  const uniqueVillages = new Set(mockWells.map((w) => w.village)).size;
  const avgConfidence =
    mockWells.reduce((sum, w) => sum + w.confidence, 0) / totalWells;

  return {
    totalWells,
    selectedArea: "Nagpur District",
    detectionAccuracy: `${Math.round(avgConfidence * 100)}%`,
    processedImages: 24, // mock static value for now
    villagesCovered: uniqueVillages,
  };
}

/**
 * Wells grouped by village/area, including average confidence per area —
 * used for the bar chart (bar color reflects avgConfidence).
 */
/**
 * Wells grouped by village/area, broken down by confidence level (High/Medium/Low)
 * — used for the stacked bar chart, using the same color scheme as the pie chart.
 */
export async function getWellsByArea() {
  await delay(MOCK_DELAY);

  const grouped = {};
  mockWells.forEach((well) => {
    if (!grouped[well.village]) {
      grouped[well.village] = { area: well.village, high: 0, medium: 0, low: 0 };
    }
    if (well.confidence >= CONFIDENCE_LEVELS.HIGH) grouped[well.village].high += 1;
    else if (well.confidence >= CONFIDENCE_LEVELS.MEDIUM) grouped[well.village].medium += 1;
    else grouped[well.village].low += 1;
  });

  return Object.values(grouped);
}

/**
 * Wells grouped by confidence level (High/Medium/Low) — used for the pie chart.
 */
export async function getConfidenceBreakdown() {
  await delay(MOCK_DELAY);

  let high = 0, medium = 0, low = 0;
  mockWells.forEach((w) => {
    if (w.confidence >= CONFIDENCE_LEVELS.HIGH) high++;
    else if (w.confidence >= CONFIDENCE_LEVELS.MEDIUM) medium++;
    else low++;
  });

  return [
    { name: "High Confidence", value: high },
    { name: "Medium Confidence", value: medium },
    { name: "Low Confidence", value: low },
  ];
}

// src/utils/leafletIconFix.js
// Vite/webpack break Leaflet's default marker icon paths.
// This file patches it so default markers still render if ever used.
import L from "leaflet";
import markerIcon2x from "leaflet/dist/images/marker-icon-2x.png";
import markerIcon from "leaflet/dist/images/marker-icon.png";
import markerShadow from "leaflet/dist/images/marker-shadow.png";

delete L.Icon.Default.prototype._getIconUrl;

L.Icon.Default.mergeOptions({
  iconRetinaUrl: markerIcon2x,
  iconUrl: markerIcon,
  shadowUrl: markerShadow,
});

// src/utils/validators.js

export function isValidEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

export function isValidMobile(mobile) {
  return /^[6-9]\d{9}$/.test(mobile); // basic 10-digit Indian mobile format
}

export function isStrongPassword(password) {
  // At least 6 characters — kept simple for a research/internal tool
  return password.length >= 6;
}
